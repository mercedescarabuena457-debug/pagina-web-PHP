2.componentes/formularioMolde.php-componente form
<?php
function renderFormularioMolde($categorias = []) {
?>
<div class="form-container">
    <h2>Crear Nuevo Molde</h2>
    <form action="/moldespro-frontend/guardarMolde.php" method="post">
        <label>Categoría:</label>
        <select name="categoria" required>
            <?php foreach($categorias as $cat): ?>
                <option value="<?php echo $cat; ?>"><?php echo $cat; ?></option>
            <?php endforeach; ?>
        </select>

        <label>Nombre de la Pieza:</label>
        <select name="nombre_pieza" required>
            <option value="Delantero">Delantero</option>
            <option value="Trasero">Trasero</option>
            <option value="Manga">Manga</option>
        </select>

        <label>Género:</label>
        <select name="genero">
            <option value="Mujer">Mujer</option>
            <option value="Hombre">Hombre</option>
        </select>

        <label>Ancho Base (cm):</label>
        <input type="number" name="ancho" step="0.1" required placeholder="Ej: 40.5">

        <label>Alto Base (cm):</label>
        <input type="number" name="alto" step="0.1" required placeholder="Ej: 60.0">

        <button type="submit">Guardar Molde</button>
    </form>
</div>
<?php 
}
?>
3.componentes/tablamoldes.php-componente tabla reutilizable
<?php
function renderTablaMoldes($moldes = []) {
?>
<div class="tabla-container">
    <h2>Últimos Moldes Registrados</h2>
    <table class="tabla">
        <thead>
            <tr>
                <th>ID</th>
                <th>Pieza</th>
                <th>Categoría</th>
                <th>Ancho</th>
                <th>Alto</th>
            </tr>
        </thead>
        <tbody>
            <?php if(empty($moldes)): ?>
                <tr><td colspan="5" class="sin-datos">No hay moldes registrados</td></tr>
            <?php else: ?>
                <?php foreach($moldes as $m): ?>
                <tr>
                    <td><?php echo $m['id']; ?></td>
                    <td><?php echo $m['nombre_pieza']; ?></td>
                    <td><?php echo $m['categoria']; ?></td>
                    <td><?php echo $m['ancho']; ?> cm</td>
                    <td><?php echo $m['alto']; ?> cm</td>
                </tr>
                <?php endforeach; ?>
            <?php endif; ?>
        </tbody>
    </table>
</div>
<?php 
}
?>
